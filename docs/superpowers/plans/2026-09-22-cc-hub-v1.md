# cc-hub v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付 PyPI 包 `cc-hub`：一个把 cc-connect 的 `work_dir` 变成"总站"的目录约定（模板）加 CLI（init / doctor / update / env / memory / jobs / hosts / hook / projects / link）。

**Architecture:** `src/cc_hub/` 是一个 typer 应用。`hub.py` 定位总站，`manifest.py` 定义框架拥有的模板文件，`encoding.py` 和 `platform.py` 收拢所有编码与跨平台差异，`commands/` 每个文件一条命令，`blocklist.py` 是钩子的黑名单。模板（Markdown、TOML、SKILL.md）随包发布，`init` 播种到用户目录，`update` 只覆盖清单内的文件。

**Tech Stack:** Python ≥ 3.11，uv，hatchling，typer，rich，tomli-w，charset-normalizer，psutil，pytest，ruff。

**Spec:** `docs/superpowers/specs/2026-09-22-cc-hub-design.md`（本计划按 spec 章节号引用，执行者两份都要读）。

## Global Constraints

- Python `>=3.11`；包名 `cc-hub`，导入名 `cc_hub`，入口命令 `cc-hub`。
- 所有 `.py` 文件首行 `# -*- coding: utf-8 -*-`，UTF-8 无 BOM，LF 换行。
- 所有文件读写显式 `encoding="utf-8"`；子进程环境加 `PYTHONUTF8=1`；路径一律 `pathlib`；启动时 `sys.stdout.reconfigure(encoding="utf-8")`。
- 来源不明的字节用 `charset-normalizer` 探测解码；发往远端的命令体一律 base64。
- json 和 toml 先写临时文件再 `os.replace`。
- 给 agent 读的命令输出纯文本，查询类命令支持 `--json`（`json.dumps(..., ensure_ascii=False, indent=2)`）。给人看的 `init`、`doctor`、`update` 用 rich。
- 退出码：成功 0，错误 1，只有 `hook pre-tool-use` 阻断用 2。
- 框架拥有的模板用英文；`AGENTS.md` ≤ 150 行；每个 `SKILL.md` ≤ 100 行。
- 仓库 `.gitattributes`：`* text=auto eol=lf`。
- 提交信息用 conventional commits（`feat:` `test:` `docs:` `chore:`），每个任务至少一次提交。
- 运行测试的命令固定为 `uv run pytest -q`；lint 为 `uv run ruff check .`。

```python
# -*- coding: utf-8 -*-
"""UTF-8 discipline: text IO, byte decoding, stdio, and base64 command wrappers."""

from __future__ import annotations

import base64
import os
import sys
import tempfile
from pathlib import Path


def configure_stdio() -> None:
    """Force UTF-8 on stdout/stderr regardless of the console's code page."""
    for stream in (sys.stdout, sys.stderr):
        reconfigure = getattr(stream, "reconfigure", None)
        if reconfigure is not None:
            reconfigure(encoding="utf-8", errors="replace")


def read_text(path: Path) -> str:
    return Path(path).read_text(encoding="utf-8")


def write_text(path: Path, text: str) -> None:
    """Atomic write: temp file in the same directory, then os.replace."""
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp_name = tempfile.mkstemp(prefix=f".{path.name}.", suffix=".tmp", dir=path.parent)
    tmp = Path(tmp_name)
    try:
        with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as fh:
            fh.write(text)
        os.replace(tmp, path)
    finally:
        if tmp.exists():
            tmp.unlink()


def append_text(path: Path, text: str) -> None:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8", newline="\n") as fh:
        fh.write(text)


def decode_bytes(data: bytes) -> str:
    """UTF-8 first; otherwise detect with charset-normalizer; never raise."""
    if not data:
        return ""
    try:
        return data.decode("utf-8")
    except UnicodeDecodeError:
        pass
    try:
        from charset_normalizer import from_bytes

        best = from_bytes(data).best()
        if best is not None:
            return str(best)
    except Exception:  # noqa: BLE001 - detection is best effort
        pass
    return data.decode("utf-8", errors="replace")


def bash_wrapper(command: str) -> str:
    """Wrap a command so a remote POSIX shell decodes and runs it without quoting issues."""
    b64 = base64.b64encode(command.encode("utf-8")).decode("ascii")
    return f'bash -c "$(printf %s {b64} | base64 -d)"'


def powershell_wrapper(command: str, exe: str = "pwsh") -> str:
    """PowerShell's -EncodedCommand expects base64 of UTF-16LE."""
    b64 = base64.b64encode(command.encode("utf-16-le")).decode("ascii")
    return f"{exe} -NoProfile -EncodedCommand {b64}"
```

```python
# -*- coding: utf-8 -*-
"""Locate the hub directory and read/write hub.toml."""

from __future__ import annotations

import os
import tomllib
from collections.abc import Mapping
from pathlib import Path

import tomli_w

from cc_hub.encoding import read_text, write_text

HUB_FILE = "hub.toml"
ENV_VAR = "CC_HUB_DIR"


class HubNotFound(Exception):
    def __init__(self) -> None:
        super().__init__(
            "No hub found. Tried: --hub, $CC_HUB_DIR, hub.toml in the current directory "
            "or any parent, and ~/.cc-hub. Run `cc-hub init` to create one."
        )


def default_hub_dir(home: Path | None = None) -> Path:
    return (home or Path.home()) / ".cc-hub"


def is_hub(path: Path) -> bool:
    return (Path(path) / HUB_FILE).is_file()


def find_hub(
    explicit: Path | None = None,
    *,
    cwd: Path | None = None,
    env: Mapping[str, str] | None = None,
    home: Path | None = None,
) -> Path:
    env = os.environ if env is None else env
    given: Path | None = None
    if explicit is not None:
        given = Path(explicit).expanduser()
    elif env.get(ENV_VAR):
        given = Path(env[ENV_VAR]).expanduser()
    if given is not None:
        if is_hub(given):
            return given.resolve()
        raise HubNotFound()
    start = (cwd or Path.cwd()).resolve()
    for candidate in (start, *start.parents):
        if is_hub(candidate):
            return candidate
    default = default_hub_dir(home)
    if is_hub(default):
        return default.resolve()
    raise HubNotFound()


def load_config(hub: Path) -> dict:
    return tomllib.loads(read_text(Path(hub) / HUB_FILE))


def save_config(hub: Path, data: Mapping) -> None:
    write_text(Path(hub) / HUB_FILE, tomli_w.dumps(dict(data)))
```

```python
# -*- coding: utf-8 -*-
"""Shared CLI helpers: hub resolution from the typer context, JSON output, rich console."""

from __future__ import annotations

import json
from pathlib import Path

import typer
from rich.console import Console

from cc_hub.hub import HubNotFound, find_hub

console = Console()


def resolve_hub(ctx: typer.Context) -> Path:
    explicit = (ctx.obj or {}).get("hub") if ctx is not None else None
    try:
        return find_hub(explicit)
    except HubNotFound as exc:
        typer.echo(str(exc), err=True)
        raise typer.Exit(code=1) from exc


def echo_json(data: object) -> None:
    typer.echo(json.dumps(data, ensure_ascii=False, indent=2, default=str))
```

```python
# -*- coding: utf-8 -*-
"""Which files the framework owns inside a hub, and how templates are rendered."""

from __future__ import annotations

from datetime import date
from pathlib import Path

from cc_hub import __version__
from cc_hub.encoding import read_text

TEMPLATES_DIR = Path(__file__).parent / "templates"

# Overwritten by `cc-hub update` (after backup).
FRAMEWORK_OWNED: tuple[str, ...] = (
    "AGENTS.md",
    "BOOTSTRAP.md",
    ".gitignore",
    "cron/README.md",
    "cron/memory-consolidate.md",
    "skills/hub-adapt/SKILL.md",
    "skills/hub-memory/SKILL.md",
    "skills/hub-jobs/SKILL.md",
    "skills/hub-hosts/SKILL.md",
    "skills/hub-projects/SKILL.md",
    "skills/hub-cc-connect/SKILL.md",
)
# Seeded once by `cc-hub init`, never touched afterwards.
USER_OWNED: tuple[str, ...] = (
    "AGENTS.local.md",
    "SOUL.md",
    "IDENTITY.md",
    "USER.md",
    "MEMORY.md",
    "projects.toml",
    "hosts.toml",
)
# Framework-owned files that `update` must not recreate once the agent removed them.
ONE_TIME: tuple[str, ...] = ("BOOTSTRAP.md",)
# Structure owned by the framework, values owned by the user: merged, never overwritten.
HUB_TOML = "hub.toml"
USER_DIRS: tuple[str, ...] = ("memory", "scripts", "secrets")
DERIVED_DIRS: tuple[str, ...] = ("jobs", "logs")

_SOURCE_NAMES = {".gitignore": "_gitignore"}


def template_source(rel: str) -> Path:
    return TEMPLATES_DIR / _SOURCE_NAMES.get(rel, rel)


def all_templates() -> tuple[str, ...]:
    return FRAMEWORK_OWNED + USER_OWNED + (HUB_TOML,)


def render(
    rel: str,
    *,
    name: str = "hub",
    version: str = __version__,
    created: date | None = None,
) -> str:
    text = read_text(template_source(rel))
    created = created or date.today()
    return (
        text.replace("{{name}}", name)
        .replace("{{version}}", version)
        .replace("{{created}}", created.isoformat())
    )
```

```python
# -*- coding: utf-8 -*-
"""cc-hub init: seed a hub directory with templates (only missing files are written)."""

from __future__ import annotations

import shutil
import subprocess
from dataclasses import dataclass, field
from pathlib import Path

import typer
from rich.table import Table

from cc_hub import manifest
from cc_hub.encoding import write_text
from cc_hub.hub import default_hub_dir


@dataclass
class InitResult:
    target: Path
    created: list[str] = field(default_factory=list)
    skipped: list[str] = field(default_factory=list)
    git_initialized: bool = False


class NestedGitError(Exception):
    pass


def enclosing_git_repo(path: Path) -> Path | None:
    path = Path(path).resolve()
    for candidate in (path, *path.parents):
        if (candidate / ".git").exists():
            return candidate
    return None


def init_hub(target: Path, *, name: str | None = None, git: bool = True) -> InitResult:
    target = Path(target).expanduser().resolve()
    repo = enclosing_git_repo(target.parent)
    if repo is not None:
        raise NestedGitError(
            f"{target} is inside the git repository {repo}. "
            "A hub must live outside every code repository."
        )
    result = InitResult(target=target)
    target.mkdir(parents=True, exist_ok=True)
    for rel in manifest.all_templates():
        dst = target / rel
        if dst.exists():
            result.skipped.append(rel)
            continue
        write_text(dst, manifest.render(rel, name=name or target.name))
        result.created.append(rel)
    for d in manifest.USER_DIRS + manifest.DERIVED_DIRS:
        (target / d).mkdir(exist_ok=True)
    if git and shutil.which("git") and not (target / ".git").exists():
        subprocess.run(["git", "init", "-q", str(target)], check=False, capture_output=True)
        result.git_initialized = (target / ".git").exists()
    return result


def register(app: typer.Typer) -> None:
    @app.command("init")
    def init_cmd(
        directory: Path | None = typer.Argument(None, help="Target directory (default: ~/.cc-hub)."),
        name: str | None = typer.Option(None, "--name", help="Hub name written to hub.toml."),
        git: bool = typer.Option(True, "--git/--no-git", help="Run `git init` in the new hub."),
    ) -> None:
        """Create a hub directory. Existing files are never overwritten."""
        from cc_hub.console import console

        target = directory or default_hub_dir()
        try:
            result = init_hub(target, name=name, git=git)
        except NestedGitError as exc:
            typer.echo(str(exc), err=True)
            raise typer.Exit(code=1) from exc
        table = Table(title=f"cc-hub init: {result.target}")
        table.add_column("file")
        table.add_column("status")
        for rel in result.created:
            table.add_row(rel, "created")
        for rel in result.skipped:
            table.add_row(rel, "kept")
        console.print(table)
        if result.git_initialized:
            console.print("Initialized a git repository in the hub.")
        console.print(
            "Next: set cc-connect's work_dir to this directory and send the first message; "
            "the agent will follow BOOTSTRAP.md."
        )
```

```python
    return "/usr/bin:/bin"
```

```python
# -*- coding: utf-8 -*-
"""cc-hub doctor: health checks for a hub directory."""

from __future__ import annotations

import re
import shutil
import time
import tomllib
from collections.abc import Callable
from dataclasses import asdict, dataclass
from pathlib import Path

import typer
from rich.table import Table

from cc_hub import __version__, manifest
from cc_hub.commands.init import enclosing_git_repo
from cc_hub.encoding import read_text, write_text
from cc_hub.hub import load_config
from cc_hub.platform import COPY_MARKER, which_minimal

Which = Callable[[str], "str | None"]


@dataclass
class Check:
    id: str
    status: str  # ok | warn | fail
    message: str
    fix: str = ""


DATE_NAME = re.compile(r"^\d{4}-\d{2}-\d{2}\.md$")
ENV_MAX_AGE_DAYS = 7
DAEMON_BINARIES = ("cc-hub", "uv", "cc-connect", "claude", "codex")
VALID_SHELLS = ("bash", "pwsh", "powershell")
AGENT_MARKERS = {"claude": ("claude", ".claude"), "codex": ("codex", ".codex")}
# agent -> [(relative path, "exists" | "contains", needle)]
POINTER_RULES: dict[str, list[tuple[str, str, str]]] = {
    "claude": [
        ("CLAUDE.md", "contains", "@AGENTS.md"),
        (".claude/skills", "exists", ""),
        (".claude/settings.json", "contains", "pre-tool-use"),
    ],
    "codex": [
        (".agents/skills", "exists", ""),
        (".codex/hooks.json", "contains", "pre-tool-use"),
    ],
}
PATH_FIX = "set PATH in cc-connect's [projects.agent.options.env] (see docs/cc-connect.md)"


def detect_agents(*, home: Path, which: Which) -> list[str]:
    return [a for a, (exe, dot) in AGENT_MARKERS.items() if which(exe) or (home / dot).exists()]


def _ok(id_: str, message: str) -> Check:
    return Check(id_, "ok", message)


def _load_table(path: Path, key: str) -> list[dict]:
    if not path.is_file():
        return []
    return list(tomllib.loads(read_text(path)).get(key, []))


def run_checks(
    hub: Path,
    *,
    fix: bool = False,
    home: Path | None = None,
    which: Which = shutil.which,
    which_daemon: Which = which_minimal,
    now: float | None = None,
    package_version: str = __version__,
) -> list[Check]:
    hub = Path(hub)
    home = home or Path.home()
    now = time.time() if now is None else now
    checks: list[Check] = []

    # hub-toml
    cfg: dict = {}
    try:
        cfg = load_config(hub)
        tv = str(cfg.get("hub", {}).get("template_version", "0"))
        if tv == package_version:
            checks.append(_ok("hub-toml", f"template_version {tv}"))
        else:
            checks.append(
                Check("hub-toml", "warn", f"template_version {tv}, package is {package_version}",
                      "run `cc-hub update` (read the migration notes in CHANGELOG.md first)")
            )
    except (OSError, tomllib.TOMLDecodeError) as exc:
        checks.append(Check("hub-toml", "fail", f"hub.toml unreadable: {exc}", "fix the TOML syntax"))
    name = str(cfg.get("hub", {}).get("name", hub.name))

    # required-files (fix: user files and .gitignore only)
    if fix:
        write_text(hub / ".gitignore", manifest.render(".gitignore", name=name, version=package_version))
        for d in manifest.USER_DIRS:
            (hub / d).mkdir(exist_ok=True)
        for rel in manifest.USER_OWNED:
            if not (hub / rel).is_file():
                write_text(hub / rel, manifest.render(rel, name=name, version=package_version))
    missing = [
        rel for rel in manifest.FRAMEWORK_OWNED + manifest.USER_OWNED
        if rel not in manifest.ONE_TIME and not (hub / rel).is_file()
    ]
    if missing:
        checks.append(Check("required-files", "fail", "missing: " + ", ".join(missing),
                            "`cc-hub doctor --fix` restores user files; `cc-hub update` restores framework files"))
    else:
        checks.append(_ok("required-files", "all present"))

    # bootstrap-pending
    if (hub / "BOOTSTRAP.md").exists():
        checks.append(Check("bootstrap-pending", "warn", "BOOTSTRAP.md still present: first-run ritual not completed",
                            "send the agent its first message"))
    else:
        checks.append(_ok("bootstrap-pending", "bootstrap completed"))

    # memory-budget
    max_lines = int(cfg.get("memory", {}).get("max_lines", 200))
    for rel, limit in (("MEMORY.md", max_lines), ("USER.md", 100)):
        p = hub / rel
        if p.is_file():
            n = len(read_text(p).splitlines())
            if n <= limit:
                checks.append(_ok(f"memory-budget:{rel}", f"{n}/{limit} lines"))
            else:
                checks.append(Check(f"memory-budget:{rel}", "fail", f"{n}/{limit} lines",
                                    "run the consolidation procedure from the hub-memory skill"))

    # memory-files
    mem = hub / "memory"
    bad = [p.name for p in mem.glob("*.md") if not DATE_NAME.match(p.name)] if mem.is_dir() else []
    if bad:
        checks.append(Check("memory-files", "warn", "unexpected names: " + ", ".join(bad), "rename to YYYY-MM-DD.md"))
    else:
        checks.append(_ok("memory-files", "names ok"))

    # projects
    try:
        projects = _load_table(hub / "projects.toml", "project")
        gone = [str(p.get("name", "?")) for p in projects
                if not Path(str(p.get("path", ""))).expanduser().is_dir()]
        if gone:
            checks.append(Check("projects", "warn", "paths missing: " + ", ".join(gone), "cc-hub projects remove <name>"))
        else:
            checks.append(_ok("projects", f"{len(projects)} registered"))
    except tomllib.TOMLDecodeError as exc:
        checks.append(Check("projects", "fail", f"projects.toml: {exc}", "fix the TOML"))

    # hosts
    try:
        hosts = _load_table(hub / "hosts.toml", "host")
        bad_shell = [str(h.get("name", "?")) for h in hosts if h.get("shell", "bash") not in VALID_SHELLS]
        if bad_shell:
            checks.append(Check("hosts", "fail", "invalid shell for: " + ", ".join(bad_shell),
                                "shell must be bash | pwsh | powershell"))
        else:
            checks.append(_ok("hosts", f"{len(hosts)} registered"))
    except tomllib.TOMLDecodeError as exc:
        checks.append(Check("hosts", "fail", f"hosts.toml: {exc}", "fix the TOML"))

    # env-stale
    env_md = hub / "ENV.md"
    if not env_md.is_file():
        checks.append(Check("env-stale", "warn", "ENV.md missing", "run `cc-hub env`"))
    else:
        age = (now - env_md.stat().st_mtime) / 86400
        if age <= ENV_MAX_AGE_DAYS:
            checks.append(_ok("env-stale", f"ENV.md is {age:.0f} days old"))
        else:
            checks.append(Check("env-stale", "warn", f"ENV.md is {age:.0f} days old", "run `cc-hub env`"))

    # pointers
    agents = detect_agents(home=home, which=which)
    if not agents:
        checks.append(_ok("pointers", "no agent runtimes detected on this machine"))
    for agent in agents:
        problems = []
        for rel, kind, needle in POINTER_RULES[agent]:
            p = hub / rel
            if kind == "exists" and not p.exists():
                problems.append(f"{rel} missing")
            elif kind == "contains" and (not p.is_file() or needle not in read_text(p)):
                problems.append(f"{rel} missing or lacks '{needle}'")
        if problems:
            checks.append(Check(f"pointers:{agent}", "warn", "; ".join(problems), "run the hub-adapt skill from the agent"))
        else:
            checks.append(_ok(f"pointers:{agent}", "pointers present"))

    # link-stale
    skills = hub / "skills"
    src_mtime = max((p.stat().st_mtime for p in skills.rglob("*") if p.is_file()), default=0.0)
    stale = [str(m.parent.relative_to(hub)) for m in hub.glob(f".*/skills/{COPY_MARKER}")
             if m.stat().st_mtime < src_mtime]
    if stale:
        checks.append(Check("link-stale", "warn", "stale copies: " + ", ".join(stale), "re-run the hub-adapt skill (cc-hub link)"))
    else:
        checks.append(_ok("link-stale", "no stale copies"))

    # daemon-path
    lost = [b for b in DAEMON_BINARIES if which(b) and not which_daemon(b)]
    if lost:
        checks.append(Check("daemon-path", "warn", "not visible under a minimal PATH: " + ", ".join(lost), PATH_FIX))
    else:
        checks.append(_ok("daemon-path", "all installed binaries visible under a minimal PATH"))

    # nested-git
    repo = enclosing_git_repo(hub.parent)
    if repo is None:
        checks.append(_ok("nested-git", "hub is outside other repositories"))
    else:
        checks.append(Check("nested-git", "fail", f"hub is inside {repo}", "move the hub outside that repository"))

    return checks
```

```python
# -*- coding: utf-8 -*-
"""cc-hub update: refresh framework-owned files, never user-owned ones."""

from __future__ import annotations

import shutil
import tomllib
from dataclasses import dataclass, field
from pathlib import Path

import typer
from rich.table import Table

from cc_hub import __version__, manifest
from cc_hub.encoding import read_text, write_text
from cc_hub.hub import load_config, save_config


@dataclass
class UpdateResult:
    updated: list[str] = field(default_factory=list)
    skipped: list[str] = field(default_factory=list)
    backup_dir: Path | None = None
    config_changed: bool = False


class UpdateError(Exception):
    pass


def merge_defaults(current: dict, defaults: dict) -> dict:
    """Return a copy of `current` with keys missing from `defaults` added, recursively."""
    out = dict(current)
    for key, value in defaults.items():
        if key not in out:
            out[key] = value
        elif isinstance(value, dict) and isinstance(out[key], dict):
            out[key] = merge_defaults(out[key], value)
    return out


def update_hub(hub: Path, *, dry_run: bool = False, version: str = __version__) -> UpdateResult:
    hub = Path(hub)
    cfg = load_config(hub)
    name = str(cfg.get("hub", {}).get("name", hub.name))
    old_version = str(cfg.get("hub", {}).get("template_version", "0"))
    result = UpdateResult()

    plan: list[tuple[str, str, bool]] = []  # (rel, new text, exists)
    for rel in manifest.FRAMEWORK_OWNED:
        dst = hub / rel
        if rel in manifest.ONE_TIME and not dst.exists():
            result.skipped.append(rel)
            continue
        new_text = manifest.render(rel, name=name, version=version)
        if dst.is_file() and read_text(dst) == new_text:
            result.skipped.append(rel)
            continue
        plan.append((rel, new_text, dst.is_file()))

    defaults = tomllib.loads(manifest.render(manifest.HUB_TOML, name=name, version=version))
    merged = merge_defaults(cfg, defaults)
    merged["hub"]["template_version"] = version
    result.config_changed = merged != cfg
    result.updated = [rel for rel, _, _ in plan]
    if dry_run:
        return result

    backup_dir = hub / ".cc-hub" / "backups" / old_version
    try:
        for rel, _, exists in plan:
            if exists:
                bak = backup_dir / rel
                bak.parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(hub / rel, bak)
    except OSError as exc:
        raise UpdateError(f"backup failed, nothing was changed: {exc}") from exc
    for rel, text, _ in plan:
        write_text(hub / rel, text)
    if result.config_changed:
        save_config(hub, merged)
    if any(exists for _, _, exists in plan):
        result.backup_dir = backup_dir
    return result


def register(app: typer.Typer) -> None:
    @app.command("update")
    def update_cmd(
        ctx: typer.Context,
        dry_run: bool = typer.Option(False, "--dry-run", help="Show what would change."),
    ) -> None:
        """Update framework-owned files to this package's templates."""
        from cc_hub.console import console, resolve_hub

        hub = resolve_hub(ctx)
        try:
            result = update_hub(hub, dry_run=dry_run)
        except UpdateError as exc:
            typer.echo(str(exc), err=True)
            raise typer.Exit(code=1) from exc
        table = Table(title=f"cc-hub update{' (dry run)' if dry_run else ''}: {hub}")
        table.add_column("file")
        table.add_column("status")
        for rel in result.updated:
            table.add_row(rel, "updated")
        for rel in result.skipped:
            table.add_row(rel, "unchanged")
        console.print(table)
        if result.backup_dir:
            console.print(f"Previous versions backed up to {result.backup_dir}")
        if result.config_changed:
            console.print("hub.toml: new keys merged, template_version updated")
```

```python
# -*- coding: utf-8 -*-
"""cc-hub env: probe this machine and write ENV.md / env.json for the agent."""

from __future__ import annotations

import json
import locale
import os
import platform as pyplatform
import shutil
import subprocess
import time
from collections.abc import Callable
from datetime import datetime
from pathlib import Path

import psutil
import typer

from cc_hub.encoding import decode_bytes, write_text
from cc_hub.hub import load_config
from cc_hub.platform import is_windows, is_wsl, which_minimal

PROBE_BINARIES = ("cc-hub", "uv", "python", "cc-connect", "claude", "codex")
SHELLS = ("bash", "zsh", "pwsh", "powershell", "cmd")
CC_CONNECT_CONFIGS = (".cc-connect/config.toml",)


def version_of(exe: str, *, run: Callable = subprocess.run) -> str:
    try:
        proc = run([exe, "--version"], capture_output=True, timeout=5)
    except (OSError, subprocess.SubprocessError):
        return ""
    lines = decode_bytes(proc.stdout or proc.stderr or b"").strip().splitlines()
    return lines[0][:80] if lines else ""


def probe(
    hub: Path,
    *,
    tools: list[str],
    home: Path | None = None,
    which: Callable[[str], str | None] = shutil.which,
    which_daemon: Callable[[str], str | None] = which_minimal,
    version_of: Callable[[str], str] = version_of,
    now: datetime | None = None,
) -> dict:
    hub = Path(hub)
    home = home or Path.home()
    now = now or datetime.now().astimezone()
    shells = {name: which(name) for name in SHELLS}
    shells["default"] = os.environ.get("SHELL") or ("powershell" if is_windows() else "sh")
    cc_cfg = next((str(home / rel) for rel in CC_CONNECT_CONFIGS if (home / rel).is_file()), None)
    return {
        "generated_at": now.isoformat(timespec="seconds"),
        "hub": str(hub),
        "system": {
            "os": pyplatform.system(),
            "release": pyplatform.release(),
            "arch": pyplatform.machine(),
            "wsl": is_wsl(),
            "home": str(home),
            "timezone": time.tzname[0],
            "locale_encoding": locale.getpreferredencoding(False),
            "pythonutf8": os.environ.get("PYTHONUTF8", ""),
        },
        "shells": shells,
        "tools": {name: {"path": which(name), "version": version_of(name) if which(name) else ""} for name in tools},
        "agents": {
            "claude_home": (home / ".claude").is_dir(),
            "codex_home": (home / ".codex").is_dir(),
            "cc_connect_config": cc_cfg,
        },
        "paths": {
            name: {"path": which(name), "daemon_visible": which_daemon(name) is not None}
            for name in PROBE_BINARIES
        },
        "resources": {
            "cpus": os.cpu_count() or 1,
            "memory_gb": round(psutil.virtual_memory().total / 2**30, 1),
            "disk_free_gb": round(shutil.disk_usage(hub).free / 2**30, 1),
        },
    }


def render_env_md(data: dict) -> str:
    s = data["system"]
    lines = [
        "# Environment",
        "",
        f"Generated by `cc-hub env` at {data['generated_at']}. Re-run when in doubt.",
        "",
        "## System",
        "",
        f"- OS: {s['os']} {s['release']} ({s['arch']}){' under WSL' if s['wsl'] else ''}",
        f"- Home: {s['home']}",
        f"- Hub: {data['hub']}",
        f"- Timezone: {s['timezone']}; locale encoding: {s['locale_encoding']}; PYTHONUTF8={s['pythonutf8'] or 'unset'}",
        "",
        "## Shell",
        "",
        f"- Default shell: {data['shells']['default']}",
    ]
    for name in SHELLS:
        path = data["shells"].get(name)
        lines.append(f"- {name}: {path or 'not installed'}")
    lines += ["", "## Tools", ""]
    for name, info in data["tools"].items():
        if info["path"]:
            lines.append(f"- {name}: {info['path']} {info['version']}".rstrip())
        else:
            lines.append(f"- {name}: not installed")
    a = data["agents"]
    lines += [
        "",
        "## Agents & bridge",
        "",
        f"- ~/.claude present: {a['claude_home']}",
        f"- ~/.codex present: {a['codex_home']}",
        f"- cc-connect config: {a['cc_connect_config'] or 'not found'}",
        "",
        "## Paths",
        "",
        "Absolute paths to use when a command is not found on PATH:",
        "",
    ]
    for name, info in data["paths"].items():
        if not info["path"]:
            lines.append(f"- {name}: not installed")
        elif info["daemon_visible"]:
            lines.append(f"- {name}: {info['path']}")
        else:
            lines.append(f"- {name}: {info['path']} (not visible to daemons with a minimal PATH)")
    r = data["resources"]
    lines += [
        "",
        "## Resources",
        "",
        f"- CPUs: {r['cpus']}; memory: {r['memory_gb']} GB; free disk at hub: {r['disk_free_gb']} GB",
        "",
    ]
    return "\n".join(lines)


def write_env(hub: Path, **probe_kwargs) -> dict:
    hub = Path(hub)
    if "tools" not in probe_kwargs:
        probe_kwargs["tools"] = list(load_config(hub).get("env", {}).get("tools", []))
    data = probe(hub, **probe_kwargs)
    write_text(hub / "env.json", json.dumps(data, ensure_ascii=False, indent=2) + "\n")
    write_text(hub / "ENV.md", render_env_md(data))
    return data


def register(app: typer.Typer) -> None:
    @app.command("env")
    def env_cmd(
        ctx: typer.Context,
        as_json: bool = typer.Option(False, "--json", help="Print env.json to stdout."),
    ) -> None:
        """Probe this machine and write ENV.md and env.json."""
        from cc_hub.console import echo_json, resolve_hub

        hub = resolve_hub(ctx)
        data = write_env(hub)
        if as_json:
            echo_json(data)
        else:
            typer.echo(f"wrote {hub / 'ENV.md'} and {hub / 'env.json'}")
```

```python
# -*- coding: utf-8 -*-
"""cc-hub memory: append-only daily logs plus search over them and MEMORY.md."""

from __future__ import annotations

import re
from dataclasses import asdict, dataclass
from datetime import date, datetime, timedelta
from pathlib import Path

import typer

from cc_hub.encoding import append_text, read_text
from cc_hub.hub import load_config

DATE_RE = re.compile(r"^(\d{4}-\d{2}-\d{2})\.md$")


@dataclass
class Hit:
    file: str
    line_no: int
    line: str
    context: str


def memory_dir(hub: Path) -> Path:
    return Path(hub) / "memory"


def day_file(hub: Path, day: date) -> Path:
    return memory_dir(hub) / f"{day.isoformat()}.md"


def log_entry(hub: Path, text: str, *, now: datetime | None = None) -> Path:
    now = now or datetime.now()
    path = day_file(hub, now.date())
    if not path.exists():
        append_text(path, f"# {now.date().isoformat()}\n\n")
    lines = text.strip().splitlines() or [""]
    body = lines[0] + "".join("\n  " + line for line in lines[1:])
    append_text(path, f"- [{now:%H:%M}] {body}\n")
    return path


def _dated_files(hub: Path, *, days: int | None, today: date) -> list[Path]:
    files: list[Path] = []
    for path in sorted(memory_dir(hub).glob("*.md"), reverse=True):
        match = DATE_RE.match(path.name)
        if not match:
            continue
        day = date.fromisoformat(match.group(1))
        if days is not None and day < today - timedelta(days=days - 1):
            continue
        files.append(path)
    return files


def recent(hub: Path, *, days: int, today: date | None = None) -> list[Path]:
    return _dated_files(hub, days=days, today=today or date.today())


def search(
    hub: Path,
    query: str,
    *,
    days: int | None = None,
    limit: int = 50,
    regex: bool = False,
    today: date | None = None,
) -> list[Hit]:
    hub = Path(hub)
    today = today or date.today()
    pattern = re.compile(query, re.IGNORECASE) if regex else None
    needle = query.casefold()
    files = [hub / "MEMORY.md", *_dated_files(hub, days=days, today=today)]
    hits: list[Hit] = []
    for path in files:
        if not path.is_file():
            continue
        lines = read_text(path).splitlines()
        for index, line in enumerate(lines):
            matched = bool(pattern.search(line)) if pattern else needle in line.casefold()
            if not matched:
                continue
            context = lines[index + 1] if index + 1 < len(lines) else ""
            hits.append(Hit(path.relative_to(hub).as_posix(), index + 1, line, context))
            if len(hits) >= limit:
                return hits
    return hits


def _count(path: Path) -> int:
    return len(read_text(path).splitlines()) if path.is_file() else 0


def stats(hub: Path, *, max_lines: int) -> dict:
    hub = Path(hub)
    logs = [p for p in memory_dir(hub).glob("*.md") if DATE_RE.match(p.name)]
    return {
        "MEMORY.md": {"lines": _count(hub / "MEMORY.md"), "max": max_lines},
        "USER.md": {"lines": _count(hub / "USER.md"), "max": 100},
        "memory/": {"files": len(logs), "lines": sum(_count(p) for p in logs)},
    }


memory_app = typer.Typer(no_args_is_help=True, help="Daily logs and long-term memory.")


@memory_app.command("log")
def log_cmd(ctx: typer.Context, text: str = typer.Argument(..., help="What to remember.")) -> None:
    """Append a timestamped entry to today's log."""
    from cc_hub.console import resolve_hub

    hub = resolve_hub(ctx)
    path = log_entry(hub, text)
    typer.echo(f"logged to {path.relative_to(hub).as_posix()}")


@memory_app.command("search")
def search_cmd(
    ctx: typer.Context,
    query: str = typer.Argument(..., help="Words or regex to look for."),
    days: int | None = typer.Option(None, "--days", help="Only logs from the last N days."),
    limit: int = typer.Option(50, "--limit"),
    regex: bool = typer.Option(False, "--regex"),
    as_json: bool = typer.Option(False, "--json"),
) -> None:
    """Search MEMORY.md and the daily logs, newest first."""
    from cc_hub.console import echo_json, resolve_hub

    hub = resolve_hub(ctx)
    hits = search(hub, query, days=days, limit=limit, regex=regex)
    if as_json:
        echo_json([asdict(h) for h in hits])
        return
    if not hits:
        typer.echo("no matches")
    for hit in hits:
        typer.echo(f"{hit.file}:{hit.line_no}: {hit.line}")


@memory_app.command("recent")
def recent_cmd(
    ctx: typer.Context,
    days: int | None = typer.Option(None, "--days", help="Default: memory.recent_days from hub.toml."),
) -> None:
    """Print the last few days of logs (what a session should read first)."""
    from cc_hub.console import resolve_hub

    hub = resolve_hub(ctx)
    days = days or int(load_config(hub).get("memory", {}).get("recent_days", 2))
    files = recent(hub, days=days)
    if not files:
        typer.echo("no memory logs yet")
        return
    for path in files:
        typer.echo(f"## {path.name}")
        typer.echo(read_text(path).rstrip())
        typer.echo("")


@memory_app.command("stats")
def stats_cmd(ctx: typer.Context, as_json: bool = typer.Option(False, "--json")) -> None:
    """Line counts against budgets."""
    from cc_hub.console import echo_json, resolve_hub

    hub = resolve_hub(ctx)
    data = stats(hub, max_lines=int(load_config(hub).get("memory", {}).get("max_lines", 200)))
    if as_json:
        echo_json(data)
        return
    for name, info in data.items():
        typer.echo(f"{name}: " + ", ".join(f"{k}={v}" for k, v in info.items()))


def register(app: typer.Typer) -> None:
    app.add_typer(memory_app, name="memory")
```

```python
# -*- coding: utf-8 -*-
"""cc-hub projects: registry of local project directories (projects.toml)."""

from __future__ import annotations

import tomllib
from datetime import date
from pathlib import Path

import tomli_w
import typer

from cc_hub.encoding import read_text, write_text

HEADER = "# Registered local projects. Managed by `cc-hub projects add|remove`.\n\n"


class ProjectError(Exception):
    pass


def load_projects(hub: Path) -> list[dict]:
    path = Path(hub) / "projects.toml"
    if not path.is_file():
        return []
    return list(tomllib.loads(read_text(path)).get("project", []))


def save_projects(hub: Path, projects: list[dict]) -> None:
    write_text(Path(hub) / "projects.toml", HEADER + tomli_w.dumps({"project": projects}))


def portable_path(path: Path, home: Path | None = None) -> str:
    resolved = Path(path).resolve()
    home = (home or Path.home()).resolve()
    try:
        return "~/" + resolved.relative_to(home).as_posix()
    except ValueError:
        return str(resolved)


def get_project(hub: Path, name: str) -> dict:
    for proj in load_projects(hub):
        if proj.get("name") == name:
            return proj
    raise ProjectError(f"no project named '{name}'")


def add_project(
    hub: Path,
    path: str | Path,
    *,
    name: str | None = None,
    description: str = "",
    notes: str = "",
    today: date | None = None,
    home: Path | None = None,
) -> dict:
    target = Path(path).expanduser().resolve()
    if not target.is_dir():
        raise ProjectError(f"{target} is not a directory")
    projects = load_projects(hub)
    for proj in projects:
        if Path(str(proj.get("path", ""))).expanduser().resolve() == target:
            raise ProjectError(f"{target} is already registered as '{proj.get('name')}'")
    name = name or target.name
    if any(p.get("name") == name for p in projects):
        raise ProjectError(f"name '{name}' is already used; pass --name")
    entry = {
        "name": name,
        "path": portable_path(target, home),
        "description": description,
        "notes": notes,
        "added": today or date.today(),
    }
    projects.append(entry)
    save_projects(hub, projects)
    return entry


def remove_project(hub: Path, name: str) -> dict:
    projects = load_projects(hub)
    keep = [p for p in projects if p.get("name") != name]
    if len(keep) == len(projects):
        raise ProjectError(f"no project named '{name}'")
    save_projects(hub, keep)
    return next(p for p in projects if p.get("name") == name)


projects_app = typer.Typer(no_args_is_help=True, help="Registry of local project directories.")


def _fail(exc: Exception) -> None:
    typer.echo(str(exc), err=True)
    raise typer.Exit(code=1)


@projects_app.command("list")
def list_cmd(ctx: typer.Context, as_json: bool = typer.Option(False, "--json")) -> None:
    """List registered projects."""
    from cc_hub.console import echo_json, resolve_hub

    projects = load_projects(resolve_hub(ctx))
    if as_json:
        echo_json(projects)
        return
    if not projects:
        typer.echo("no projects registered; use `cc-hub projects add <path>`")
        return
    for p in projects:
        desc = f"  {p.get('description')}" if p.get("description") else ""
        typer.echo(f"{p.get('name')}\t{p.get('path')}{desc}")


@projects_app.command("show")
def show_cmd(ctx: typer.Context, name: str, as_json: bool = typer.Option(False, "--json")) -> None:
    """Show one project."""
    from cc_hub.console import echo_json, resolve_hub

    try:
        proj = get_project(resolve_hub(ctx), name)
    except ProjectError as exc:
        _fail(exc)
    if as_json:
        echo_json(proj)
        return
    for key, value in proj.items():
        typer.echo(f"{key}: {value}")


@projects_app.command("add")
def add_cmd(
    ctx: typer.Context,
    path: Path = typer.Argument(..., help="Project directory."),
    name: str | None = typer.Option(None, "--name"),
    description: str = typer.Option("", "--description"),
    notes: str = typer.Option("", "--notes", help="How to run/test it, quirks."),
) -> None:
    """Register a project directory."""
    from cc_hub.console import resolve_hub

    try:
        entry = add_project(resolve_hub(ctx), path, name=name, description=description, notes=notes)
    except ProjectError as exc:
        _fail(exc)
    typer.echo(f"registered {entry['name']} -> {entry['path']}")


@projects_app.command("remove")
def remove_cmd(ctx: typer.Context, name: str) -> None:
    """Remove a project from the registry (the directory is untouched)."""
    from cc_hub.console import resolve_hub

    try:
        remove_project(resolve_hub(ctx), name)
    except ProjectError as exc:
        _fail(exc)
    typer.echo(f"removed {name}")


def register(app: typer.Typer) -> None:
    app.add_typer(projects_app, name="projects")
```

```python
class LinkError(Exception):
    pass


def link_dir(
    src: Path,
    dst: Path,
    *,
    windows: bool | None = None,
    run=subprocess.run,
) -> str:
    """Make `dst` reach `src`: symlink on POSIX, junction on Windows, copy as last resort."""
    src = Path(src).resolve()
    dst = Path(dst)
    if not src.is_dir():
        raise LinkError(f"{src} is not a directory")
    dst.parent.mkdir(parents=True, exist_ok=True)
    if dst.is_symlink():
        if dst.resolve() == src:
            return "existing"
        dst.unlink()
    elif dst.exists():
        if dst.resolve() == src:
            return "existing"
        if (dst / COPY_MARKER).is_file():
            shutil.rmtree(dst)
        else:
            raise LinkError(f"{dst} exists and was not created by cc-hub; remove it first")
    windows = is_windows() if windows is None else windows
    if not windows:
        dst.symlink_to(src, target_is_directory=True)
        return "symlink"
    try:
        run(["cmd", "/c", "mklink", "/J", str(dst), str(src)], check=True, capture_output=True)
        return "junction"
    except (OSError, subprocess.CalledProcessError):
        shutil.copytree(src, dst)
        (dst / COPY_MARKER).write_text(f"copied from {src}\n", encoding="utf-8")
        return "copy"
```

```python
# -*- coding: utf-8 -*-
"""Catastrophic-command blocklist used by `cc-hub hook pre-tool-use`."""

from __future__ import annotations

import re
import shlex
from collections.abc import Callable, Iterable
from dataclasses import dataclass
from pathlib import Path


@dataclass
class Segment:
    raw: str
    name: str
    args: list[str]


@dataclass(frozen=True)
class Rule:
    name: str
    scope: str  # "segment" | "command"
    test: Callable[..., bool]


SPLIT_RE = re.compile(r"\|\||&&|;|\||\n")
WRAPPERS = {"sudo", "doas", "nohup", "time", "command", "exec", "builtin", "env", "nice", "timeout"}
WRAPPER_OPTS_WITH_ARG = {"-u", "-g", "-p", "-n"}
ROOT_TARGETS = {
    "/", "/*", "~", "~/", "~/*", "$home", "$home/", "${home}", "${home}/", "/home", "/home/*",
    "$env:userprofile", "$env:userprofile\\", "$env:systemdrive", "$env:systemdrive\\",
}
ROOT_TARGET_RE = re.compile(r"^(?:[a-z]:[\\/]?\*?|/home/[^/]+/?|/users/[^/]+/?|c:\\users\\[^\\]+\\?)$")
DEVICE_REDIRECT = re.compile(r">\s*/dev/(?:sd|nvme|disk|hd|mmcblk|xvd)")
FORK_BOMB = re.compile(r":\s*\(\s*\)\s*\{\s*:\s*\|\s*:\s*&\s*\}\s*;\s*:")
PIPE_TO_SHELL = re.compile(r"\b(?:curl|wget)\b[^|]*\|\s*(?:sudo\s+(?:-\S+\s+)*)?(?:bash|sh|zsh|dash|fish)\b")
POWER = {"shutdown", "reboot", "poweroff", "halt", "stop-computer", "restart-computer"}
WIN_REMOVE = {"remove-item", "ri", "del", "erase", "rd", "rmdir"}
FORMAT = {"format", "format.com", "diskpart", "diskpart.exe"}
MAIN_REF = re.compile(r"^(?:.*:)?(?:refs/heads/)?(?:main|master)$")
PROTECTED_NAME = re.compile(r"^\.env(?:\..+)?$")


def normalize(command: str) -> str:
    return " ".join(command.split())


def split_segments(command: str) -> list[Segment]:
    segments: list[Segment] = []
    for part in SPLIT_RE.split(command):
        part = part.strip()
        if not part:
            continue
        try:
            tokens = shlex.split(part, posix=True)
        except ValueError:
            tokens = part.split()
        while tokens and (tokens[0] in WRAPPERS or ("=" in tokens[0] and not tokens[0].startswith("-"))):
            head = tokens.pop(0)
            if head in ("sudo", "doas", "timeout", "nice", "env"):
                while tokens and tokens[0].startswith("-"):
                    opt = tokens.pop(0)
                    if opt in WRAPPER_OPTS_WITH_ARG and tokens:
                        tokens.pop(0)
                if head == "timeout" and tokens:
                    tokens.pop(0)  # the duration
        if not tokens:
            continue
        name = Path(tokens[0]).name.casefold()
        segments.append(Segment(raw=part, name=name, args=tokens[1:]))
    return segments


def _short_flags(args: list[str]) -> str:
    return "".join(a[1:] for a in args if a.startswith("-") and not a.startswith("--") and len(a) > 1)


def _recursive(args: list[str]) -> bool:
    if "--recursive" in args:
        return True
    lowered = [a.casefold() for a in args]
    if any(a in ("-recurse", "/s") or a.startswith("-recurse") for a in lowered):
        return True
    return any(ch in "rR" for ch in _short_flags(args))


def _targets(args: list[str]) -> list[str]:
    return [a for a in args if not a.startswith("-") and a.casefold() not in ("/s", "/q")]


def _is_root_target(target: str) -> bool:
    t = target.casefold()
    return t in ROOT_TARGETS or ROOT_TARGET_RE.match(t) is not None


def rule_rm_root(seg: Segment) -> bool:
    return seg.name == "rm" and _recursive(seg.args) and any(_is_root_target(t) for t in _targets(seg.args))


def rule_mkfs(seg: Segment) -> bool:
    return seg.name.startswith("mkfs")


def rule_dd_device(seg: Segment) -> bool:
    return seg.name == "dd" and any(a.startswith("of=/dev/") for a in seg.args)


def rule_power(seg: Segment) -> bool:
    if seg.name in POWER:
        return True
    return seg.name == "systemctl" and any(a in ("poweroff", "reboot", "halt") for a in seg.args)


def rule_force_push_main(seg: Segment) -> bool:
    if seg.name != "git" or "push" not in seg.args:
        return False
    force = any(a in ("-f", "--force") for a in seg.args)
    return force and any(MAIN_REF.match(a) for a in seg.args if not a.startswith("-"))


def rule_chmod_root(seg: Segment) -> bool:
    return seg.name in ("chmod", "chown", "chgrp") and _recursive(seg.args) and any(
        _is_root_target(t) for t in _targets(seg.args)
    )


def rule_win_remove_root(seg: Segment) -> bool:
    return seg.name in WIN_REMOVE and _recursive(seg.args) and any(
        _is_root_target(t) for t in _targets(seg.args)
    )


def rule_win_format(seg: Segment) -> bool:
    return seg.name in FORMAT


BUILTIN_RULES: tuple[Rule, ...] = (
    Rule("rm-root", "segment", rule_rm_root),
    Rule("mkfs", "segment", rule_mkfs),
    Rule("dd-device", "segment", rule_dd_device),
    Rule("redirect-device", "command", lambda text: DEVICE_REDIRECT.search(text) is not None),
    Rule("fork-bomb", "command", lambda text: FORK_BOMB.search(text) is not None),
    Rule("pipe-to-shell", "command", lambda text: PIPE_TO_SHELL.search(text) is not None),
    Rule("power", "segment", rule_power),
    Rule("force-push-main", "segment", rule_force_push_main),
    Rule("chmod-root", "segment", rule_chmod_root),
    Rule("win-remove-root", "segment", rule_win_remove_root),
    Rule("win-format", "segment", rule_win_format),
)


def match_command(command: str, extra: Iterable[str] = ()) -> str | None:
    text = normalize(command)
    if not text:
        return None
    segments = split_segments(text)
    for rule in BUILTIN_RULES:
        if rule.scope == "command":
            if rule.test(text):
                return rule.name
        elif any(rule.test(seg) for seg in segments):
            return rule.name
    for index, pattern in enumerate(extra):
        try:
            if re.search(pattern, text, re.IGNORECASE):
                return f"extra:{index}"
        except re.error:
            continue
    return None


def is_protected_path(path: str, hub: Path) -> bool:
    candidate = Path(path).expanduser()
    if not candidate.is_absolute():
```

```python
"""cc-hub hook: pre/post tool-use hooks. Same JSON protocol for Claude Code and Codex."""

from __future__ import annotations

import json
import shutil
import sys
from collections.abc import Iterable
from datetime import datetime
from pathlib import Path

import typer

from cc_hub.hub import HubNotFound, find_hub, load_config

COMMAND_KEYS = ("command", "cmd")
PATH_KEYS = ("file_path", "path", "filePath")


def extract_command(tool_input: dict) -> str | None:
    for key in COMMAND_KEYS:
        value = tool_input.get(key)
        if isinstance(value, str) and value.strip():
            return value
        if isinstance(value, list) and value:
            return " ".join(str(v) for v in value)
    return None


def extract_path(tool_input: dict) -> str | None:
    for key in PATH_KEYS:
        value = tool_input.get(key)
        if isinstance(value, str) and value.strip():
            return value
    return None


def evaluate(payload: dict, hub: Path, *, extra: Iterable[str] = ()) -> tuple[bool, str]:
    tool_input = payload.get("tool_input")
    if not isinstance(tool_input, dict):
        return False, ""
    command = extract_command(tool_input)
    if command:
        rule = match_command(command, extra)
        if rule:
            return True, f"cc-hub blocked this command (rule {rule}): {normalize(command)[:200]}"
    path = extract_path(tool_input)
    if path and is_protected_path(path, hub):
        return True, f"cc-hub blocked access to a protected path (secrets/ or .env): {path}"
    return False, ""


def default_exe() -> str:
    found = shutil.which("cc-hub")
    return found if found else f"{sys.executable} -m cc_hub"


def snippet(exe: str | None = None, *, matcher: bool = True) -> dict:
    exe = exe or default_exe()
    pre: dict = {}
    post: dict = {}
    if matcher:
        pre["matcher"] = "Bash|PowerShell|Read"
        post["matcher"] = "Bash|PowerShell"
    pre["hooks"] = [{"type": "command", "command": f"{exe} hook pre-tool-use"}]
    post["hooks"] = [{"type": "command", "command": f"{exe} hook post-tool-use", "async": True}]
    return {"hooks": {"PreToolUse": [pre], "PostToolUse": [post]}}


def audit_record(payload: dict, *, now: datetime | None = None) -> dict:
    now = now or datetime.now().astimezone()
    tool_input = payload.get("tool_input")
    tool_input = tool_input if isinstance(tool_input, dict) else {}
    response = payload.get("tool_response")
    record: dict = {
        "ts": now.isoformat(timespec="seconds"),
        "session_id": payload.get("session_id"),
        "tool": payload.get("tool_name"),
        "cwd": payload.get("cwd"),
    }
    command = extract_command(tool_input)
    if command:
        record["command"] = normalize(command)[:500]
    path = extract_path(tool_input)
    if path:
        record["path"] = path
    exit_code = None
    if isinstance(response, dict):
        exit_code = response.get("exit_code", response.get("exitCode"))
        record["success"] = not response.get("is_error", False) and exit_code in (None, 0)
    else:
        record["success"] = not payload.get("error")
    if exit_code is not None:
        record["exit_code"] = exit_code
    return record


def append_audit(hub: Path, record: dict) -> Path:
    path = Path(hub) / "logs" / f"{record['ts'][:10]}.jsonl"
    append_text(path, json.dumps(record, ensure_ascii=False) + "\n")
    return path


def read_payload(stream=None) -> dict:
    stream = stream or sys.stdin
    raw = stream.buffer.read() if hasattr(stream, "buffer") else stream.read().encode("utf-8")
    data = json.loads(decode_bytes(raw))
    if not isinstance(data, dict):
        raise ValueError("payload is not a JSON object")
    return data


def _try_hub(ctx: typer.Context) -> Path | None:
    try:
        return find_hub((ctx.obj or {}).get("hub"))
    except HubNotFound:
        return None


def _extra(hub: Path | None) -> list[str]:
    if hub is None:
        return []
    try:
        return [str(p) for p in load_config(hub).get("hooks", {}).get("blocklist_extra", [])]
    except Exception:  # noqa: BLE001 - a broken hub.toml must not disable the built-in rules
        return []


hook_app = typer.Typer(no_args_is_help=True, help="Hooks invoked by the agent runtime around tool calls.")


@hook_app.command("pre-tool-use")
def pre_tool_use(ctx: typer.Context) -> None:
    """Read the tool-call JSON from stdin; exit 2 to block catastrophic commands."""
    hub = _try_hub(ctx)
    try:
        payload = read_payload()
    except (ValueError, OSError) as exc:
        typer.echo(f"cc-hub hook: could not parse payload ({exc}); allowing", err=True)
        raise typer.Exit(code=0) from exc
    blocked, reason = evaluate(payload, hub or Path.cwd(), extra=_extra(hub))
    if blocked:
        typer.echo(reason, err=True)
        raise typer.Exit(code=2)


@hook_app.command("post-tool-use")
def post_tool_use(ctx: typer.Context) -> None:
    """Append one JSON line per tool call to logs/YYYY-MM-DD.jsonl. Never fails."""
    try:
        hub = _try_hub(ctx)
        if hub is None:
            return
        append_audit(hub, audit_record(read_payload()))
    except Exception as exc:  # noqa: BLE001 - logging must never block the agent
        typer.echo(f"cc-hub hook: audit skipped ({exc})", err=True)


@hook_app.command("test")
def test_cmd(ctx: typer.Context, command: str = typer.Argument(..., help="Command to check.")) -> None:
    """Check a command against the blocklist without running it."""
    rule = match_command(command, _extra(_try_hub(ctx)))
    if rule:
        typer.echo(f"BLOCKED by rule {rule}")
        raise typer.Exit(code=2)
    typer.echo("allowed")


@hook_app.command("snippet")
def snippet_cmd(
    exe: str | None = typer.Option(None, "--exe", help="Absolute path to cc-hub (default: auto-detect)."),
    no_matcher: bool = typer.Option(False, "--no-matcher", help="Match every tool; filter inside the hook."),
) -> None:
    """Print the hooks JSON to merge into the agent's hook configuration."""
    from cc_hub.console import echo_json

    echo_json(snippet(exe, matcher=not no_matcher))


def register(app: typer.Typer) -> None:
    app.add_typer(hook_app, name="hook")
```

```python
def child_env(extra: dict[str, str] | None = None) -> dict[str, str]:
    env = dict(os.environ)
    env["PYTHONUTF8"] = "1"
    env.update(extra or {})
    return env


def spawn_detached(argv: list[str], *, cwd: Path, env: dict[str, str] | None = None) -> int:
    """Start argv detached from this process and its session; return the pid."""
    kwargs: dict = {
        "cwd": str(cwd),
        "env": env or child_env(),
        "stdin": subprocess.DEVNULL,
        "stdout": subprocess.DEVNULL,
        "stderr": subprocess.DEVNULL,
        "close_fds": True,
    }
    if is_windows():
        kwargs["creationflags"] = subprocess.CREATE_NEW_PROCESS_GROUP | subprocess.DETACHED_PROCESS
    else:
        kwargs["start_new_session"] = True
    return subprocess.Popen(argv, **kwargs).pid


def pid_alive(pid: int | None) -> bool:
    if not pid or not psutil.pid_exists(pid):
        return False
    try:
        return psutil.Process(pid).status() != psutil.STATUS_ZOMBIE
    except psutil.Error:
        return False


def kill_tree(pid: int, *, force: bool = False) -> None:
    """Terminate a process and everything under it."""
    if is_windows():
        argv = ["taskkill", "/T", "/PID", str(pid)] + (["/F"] if force else [])
        subprocess.run(argv, capture_output=True, check=False)
        return
    sig = signal.SIGKILL if force else signal.SIGTERM
    try:
        os.killpg(os.getpgid(pid), sig)
    except (ProcessLookupError, PermissionError):
        pass
```

```python
# -*- coding: utf-8 -*-
"""cc-hub jobs: detached background jobs supervised by a runner process."""

from __future__ import annotations

import json
import os
import re
import secrets
import subprocess
import sys
import time
from dataclasses import asdict, dataclass
from datetime import datetime, timedelta
from pathlib import Path

import typer

from cc_hub.encoding import decode_bytes, read_text, write_text
from cc_hub.platform import child_env, kill_tree, pid_alive, spawn_detached

STATUSES = ("running", "done", "failed", "stopped", "lost")
STARTUP_GRACE = timedelta(seconds=10)


class JobError(Exception):
    pass


@dataclass
class Job:
    id: str
    name: str
    cmd: list[str]
    cwd: str
    pid: int | None
    runner_pid: int | None
    started: str
    ended: str | None
    status: str
    exit_code: int | None
    log: str


def jobs_dir(hub: Path) -> Path:
    return Path(hub) / "jobs"


def job_path(hub: Path, job_id: str) -> Path:
    return jobs_dir(hub) / f"{job_id}.json"


def _now() -> str:
    return datetime.now().astimezone().isoformat(timespec="seconds")


def new_id(now: datetime | None = None) -> str:
    now = now or datetime.now()
    return f"{now:%Y%m%d-%H%M%S}-{secrets.token_hex(2)}"


def save_job(hub: Path, job: Job) -> None:
    write_text(job_path(hub, job.id), json.dumps(asdict(job), ensure_ascii=False, indent=2) + "\n")


def load_job(hub: Path, job_id: str) -> Job:
    path = job_path(hub, job_id)
    if not path.is_file():
        raise JobError(f"no job named {job_id}")
    return Job(**json.loads(read_text(path)))


def list_jobs(hub: Path) -> list[Job]:
    return [Job(**json.loads(read_text(p))) for p in sorted(jobs_dir(hub).glob("*.json"), reverse=True)]


def start_job(hub: Path, cmd: list[str], *, cwd: Path, name: str | None = None) -> Job:
    hub = Path(hub)
    if not cmd:
        raise JobError("no command given (put it after --)")
    cwd = Path(cwd).expanduser().resolve()
    if not cwd.is_dir():
        raise JobError(f"{cwd} is not a directory")
    jobs_dir(hub).mkdir(exist_ok=True)
    job_id = new_id()
    job = Job(
        id=job_id, name=name or Path(cmd[0]).name, cmd=list(cmd), cwd=str(cwd), pid=None, runner_pid=None,
        started=_now(), ended=None, status="running", exit_code=None, log=f"jobs/{job_id}.log",
    )
    save_job(hub, job)
    runner = [sys.executable, "-m", "cc_hub", "--hub", str(hub), "jobs", "_run", job_id]
    job.runner_pid = spawn_detached(runner, cwd=hub)
    # Do not save here: the runner writes runner_pid/pid itself; a second write would race it.
    return job


def run_job(hub: Path, job_id: str) -> int:
    """Runner body: execute the command, stream output to the log, record the outcome."""
    hub = Path(hub)
    job = load_job(hub, job_id)
    job.runner_pid = os.getpid()
    save_job(hub, job)
    log = hub / job.log
    log.parent.mkdir(exist_ok=True)
    with log.open("ab") as fh:
        try:
            proc = subprocess.Popen(
                job.cmd, cwd=job.cwd, stdin=subprocess.DEVNULL, stdout=fh, stderr=subprocess.STDOUT, env=child_env()
            )
        except OSError as exc:
            fh.write(f"cc-hub: cannot start command: {exc}\n".encode("utf-8"))
            job.status, job.exit_code, job.ended = "failed", 127, _now()
            save_job(hub, job)
            return 127
        job.pid = proc.pid
        save_job(hub, job)
        code = proc.wait()
    job.exit_code = code
    job.ended = _now()
    job.status = "done" if code == 0 else "failed"
    save_job(hub, job)
    return code


def is_lost(job: Job) -> bool:
    if job.status != "running":
        return False
    if job.runner_pid is None:
        started = datetime.fromisoformat(job.started)
        if datetime.now(started.tzinfo) - started < STARTUP_GRACE:
            return False
    return not (pid_alive(job.runner_pid) or pid_alive(job.pid))


def reconcile(hub: Path, job: Job) -> Job:
    if is_lost(job):
        job.status = "lost"
        job.ended = _now()
        save_job(hub, job)
    return job


def stop_job(hub: Path, job_id: str, *, force: bool = False, grace: float = 10.0) -> Job:
    job = reconcile(hub, load_job(hub, job_id))
    if job.status != "running":
        return job
    target = job.runner_pid or job.pid
    if target:
        kill_tree(target, force=force)
    deadline = time.monotonic() + grace
    while time.monotonic() < deadline and (pid_alive(job.runner_pid) or pid_alive(job.pid)):
        time.sleep(0.1)
    if target and (pid_alive(job.runner_pid) or pid_alive(job.pid)):
        kill_tree(target, force=True)
        time.sleep(0.3)
    job = load_job(hub, job_id)
    if job.status == "running":
        job.status = "stopped"
        job.ended = _now()
        save_job(hub, job)
    return job


def tail(hub: Path, job_id: str, n: int) -> str:
    path = Path(hub) / load_job(hub, job_id).log
    if not path.is_file():
        return ""
    return "\n".join(decode_bytes(path.read_bytes()).splitlines()[-n:])


def parse_days(text: str) -> int:
    match = re.fullmatch(r"(\d+)d?", text.strip())
    if not match:
        raise ValueError(f"expected a number of days like 7d, got {text!r}")
    return int(match.group(1))


def clean(hub: Path, *, older_than_days: int) -> list[str]:
    cutoff = datetime.now().astimezone() - timedelta(days=older_than_days)
    removed: list[str] = []
    for job in list_jobs(hub):
        job = reconcile(hub, job)
        if job.status == "running" or not job.ended:
            continue
        ended = datetime.fromisoformat(job.ended)
        if ended.tzinfo is None:
            ended = ended.astimezone()
        if ended < cutoff:
            job_path(hub, job.id).unlink(missing_ok=True)
            (Path(hub) / job.log).unlink(missing_ok=True)
            removed.append(job.id)
    return removed
```

```python
# -*- coding: utf-8 -*-
"""cc-hub hosts: run commands and copy files on registered remote machines over ssh."""

from __future__ import annotations

import shlex
import subprocess
import tomllib
from collections.abc import Callable
from pathlib import Path

import typer

from cc_hub.encoding import bash_wrapper, decode_bytes, powershell_wrapper, read_text

SHELLS = ("bash", "pwsh", "powershell")


class HostError(Exception):
    pass


def load_hosts(hub: Path) -> list[dict]:
    path = Path(hub) / "hosts.toml"
    if not path.is_file():
        return []
    return list(tomllib.loads(read_text(path)).get("host", []))


def get_host(hub: Path, name: str) -> dict:
    for host in load_hosts(hub):
        if host.get("name") == name:
            if host.get("shell", "bash") not in SHELLS:
                raise HostError(f"host {name}: shell must be one of {', '.join(SHELLS)}")
            if not host.get("target"):
                raise HostError(f"host {name}: target is required")
            return host
    raise HostError(f"no host named '{name}' in hosts.toml")


def build_remote_command(host: dict, command: str) -> str:
    shell = host.get("shell", "bash")
    if shell == "bash":
        return bash_wrapper(command)
    return powershell_wrapper(command, exe=shell)


def _common_opts(host: dict, hub: Path, *, port_flag: str) -> list[str]:
    opts = ["-o", "BatchMode=yes"]
    if host.get("port"):
        opts += [port_flag, str(host["port"])]
    if host.get("key"):
        key = Path(str(host["key"])).expanduser()
        if not key.is_absolute():
            key = Path(hub) / key
        opts += ["-i", str(key)]
    return opts


def ssh_argv(host: dict, hub: Path, remote_command: str | None) -> list[str]:
    argv = ["ssh", *_common_opts(host, hub, port_flag="-p"), str(host["target"])]
    if remote_command:
        argv.append(remote_command)
    return argv


def scp_argv(host: dict, hub: Path, src: str, dst: str, *, upload: bool) -> list[str]:
    target = str(host["target"])
    pair = [src, f"{target}:{dst}"] if upload else [f"{target}:{src}", dst]
    return ["scp", *_common_opts(host, hub, port_flag="-P"), *pair]


def run_ssh(argv: list[str], *, timeout: float | None = None, run: Callable = subprocess.run) -> tuple[int, str, str]:
    try:
        proc = run(argv, capture_output=True, timeout=timeout)
    except FileNotFoundError as exc:
        raise HostError(f"{argv[0]} not found; install OpenSSH client") from exc
    except subprocess.TimeoutExpired as exc:
        raise HostError(f"{argv[0]} timed out after {timeout}s") from exc
    return proc.returncode, decode_bytes(proc.stdout or b""), decode_bytes(proc.stderr or b"")


def run_remote(hub: Path, name: str, command: str, *, timeout: float | None = None, run: Callable = subprocess.run) -> tuple[int, str, str]:
    host = get_host(hub, name)
    return run_ssh(ssh_argv(host, hub, build_remote_command(host, command)), timeout=timeout, run=run)


def check_host(hub: Path, name: str, *, run: Callable = subprocess.run) -> tuple[int, str, str]:
    host = get_host(hub, name)
    return run_ssh(ssh_argv(host, hub, "exit 0"), timeout=30, run=run)


def join_command(parts: list[str]) -> str:
    """One argument is taken verbatim (may contain && and quotes); several are shell-joined."""
    return parts[0] if len(parts) == 1 else shlex.join(parts)


hosts_app = typer.Typer(no_args_is_help=True, help="Remote hosts over ssh (keys only, no passwords).")
_RAW = {"allow_interspersed_args": False, "ignore_unknown_options": True}


def _fail(exc: Exception) -> None:
    typer.echo(str(exc), err=True)
    raise typer.Exit(code=1)


def _emit(code: int, out: str, err: str) -> None:
    if out:
        typer.echo(out, nl=False)
    if err:
        typer.echo(err, err=True, nl=False)
    raise typer.Exit(code=code)


@hosts_app.command("list")
def list_cmd(ctx: typer.Context, as_json: bool = typer.Option(False, "--json")) -> None:
    """List registered hosts."""
    from cc_hub.console import echo_json, resolve_hub

    hosts = load_hosts(resolve_hub(ctx))
    if as_json:
        echo_json(hosts)
        return
    if not hosts:
        typer.echo("no hosts registered; add a [[host]] block to hosts.toml")
        return
    for h in hosts:
        notes = f"  {h.get('notes')}" if h.get("notes") else ""
        typer.echo(f"{h.get('name')}\t{h.get('target')}\t{h.get('shell', 'bash')}{notes}")


@hosts_app.command("check")
def check_cmd(ctx: typer.Context, name: str) -> None:
    """Verify key-based login works."""
    from cc_hub.console import resolve_hub

    try:
        code, _, err = check_host(resolve_hub(ctx), name)
    except HostError as exc:
        _fail(exc)
    typer.echo("ok" if code == 0 else f"failed ({code}): {err.strip()}")
    raise typer.Exit(code=code)


@hosts_app.command("run", context_settings=_RAW)
def run_cmd(
    ctx: typer.Context,
    name: str,
    command: list[str] = typer.Argument(..., help="Remote command; put -- before it."),
    timeout: float | None = typer.Option(None, "--timeout", help="Seconds."),
) -> None:
    """Run a command on the host; output and exit code are passed through."""
    from cc_hub.console import resolve_hub

    try:
        code, out, err = run_remote(resolve_hub(ctx), name, join_command(command), timeout=timeout)
    except HostError as exc:
        _fail(exc)
    _emit(code, out, err)


@hosts_app.command("push")
def push_cmd(ctx: typer.Context, name: str, local: Path, remote: str) -> None:
    """Copy a local file to the host (scp)."""
    from cc_hub.console import resolve_hub

    hub = resolve_hub(ctx)
    try:
        host = get_host(hub, name)
        code, out, err = run_ssh(scp_argv(host, hub, str(local), remote, upload=True))
    except HostError as exc:
        _fail(exc)
    _emit(code, out, err)


@hosts_app.command("pull")
def pull_cmd(ctx: typer.Context, name: str, remote: str, local: Path) -> None:
    """Copy a file from the host (scp)."""
    from cc_hub.console import resolve_hub

    hub = resolve_hub(ctx)
    try:
        host = get_host(hub, name)
        code, out, err = run_ssh(scp_argv(host, hub, remote, str(local), upload=False))
    except HostError as exc:
        _fail(exc)
    _emit(code, out, err)


def register(app: typer.Typer) -> None:
    app.add_typer(hosts_app, name="hosts")
```

```markdown
The hub is agent-agnostic. It ships no Claude- or Codex-specific files; on
first run the agent creates the pointers its own ecosystem needs (`hub-adapt`).

## Quick start

    uv tool install cc-hub
    cc-hub init                     # creates ~/.cc-hub
    cc-hub doctor

Point cc-connect at the hub (`work_dir = "/home/you/.cc-hub"`, see
[docs/cc-connect.md](docs/cc-connect.md)) and send the first message. The agent
follows `BOOTSTRAP.md`: names itself, learns who you are, registers projects,
adapts to its runtime, and deletes the bootstrap file.

## Commands

| Command | Purpose |
|---|---|
| `cc-hub init [dir]` | Seed a hub. Existing files are never overwritten. |
| `cc-hub doctor [--fix] [--json]` | Health checks: budgets, pointers, daemon PATH, orphaned jobs. |
| `cc-hub update [--dry-run]` | Refresh framework-owned files; user files untouched; backups kept. |
| `cc-hub env` | Probe the machine and write `ENV.md` / `env.json`. |
| `cc-hub memory log\|search\|recent\|stats` | Daily logs and long-term memory. |
| `cc-hub projects list\|show\|add\|remove` | Registry of local project directories. |
| `cc-hub jobs start\|status\|log\|stop\|clean` | Detached background jobs that outlive the chat session. |
| `cc-hub hosts list\|check\|run\|push\|pull` | Remote machines over ssh; commands are base64-wrapped. |
| `cc-hub hook pre-tool-use\|post-tool-use\|test\|snippet` | Blocklist and audit log hooks for the agent runtime. |
| `cc-hub link <src> <dst>` | Symlink, junction or copy (platform decides). |

## Hub layout

    ~/.cc-hub/
    ├── AGENTS.md  AGENTS.local.md  BOOTSTRAP.md      instructions
    ├── SOUL.md  IDENTITY.md  USER.md                 persona and user
    ├── MEMORY.md  memory/YYYY-MM-DD.md               memory
    ├── projects.toml  hosts.toml  hub.toml           registries and config
    ├── skills/  cron/  scripts/  secrets/            skills, scheduled prompts, your scripts, credentials
    └── ENV.md  jobs/  logs/                          generated state (gitignored)

Framework-owned files (`AGENTS.md`, `BOOTSTRAP.md`, `skills/hub-*`, `cron/README.md`)
are replaced by `cc-hub update`. Everything else is yours.
```

```markdown
# Install and upgrade

## Requirements

- [uv](https://docs.astral.sh/uv/) (single binary; installs Python 3.11+ itself if needed)
- OpenSSH client if you use `cc-hub hosts`
- git (optional; `cc-hub init` initializes a repository in the hub)

## Install

    uv tool install cc-hub
    cc-hub version

`uv tool install` puts `cc-hub` in `~/.local/bin` (Windows: `%USERPROFILE%\.local\bin`).
Make sure that directory is on your PATH; `uv tool update-shell` can add it.

## Create the hub

    cc-hub init              # ~/.cc-hub
    cc-hub init ~/my-hub     # anywhere else

The hub must not live inside another git repository; `init` refuses that.
Treat the hub as private data: commit it to a private repository if you want
backups. `secrets/`, generated files and agent pointers are already gitignored.

## Connect cc-connect

See [cc-connect.md](cc-connect.md). Then send the agent its first message.

## Check

    cc-hub doctor

Warnings about `bootstrap-pending` and `pointers` disappear after the first
conversation. A `daemon-path` warning means cc-connect, when run as a service,
would not find `cc-hub` or `uv`; fix it in cc-connect's config as described in
[cc-connect.md](cc-connect.md).

## Upgrade

    uv tool upgrade cc-hub
    cc-hub update --dry-run   # see what would change
    cc-hub update

`update` replaces only framework-owned files (`AGENTS.md`, `BOOTSTRAP.md`,
`skills/hub-*`, `cron/README.md`, `cron/memory-consolidate.md`, `.gitignore`)
and merges new keys into `hub.toml`. Previous versions are copied to
`.cc-hub/backups/<old version>/`. Read the migration notes in `CHANGELOG.md`
before upgrading across minor versions.
```

`docs/cc-connect.md`:

```markdown
# cc-connect configuration

Minimal `[[projects]]` entry pointing a Claude Code bot at the hub:

    [[projects]]
    name = "hub"
    admin_from = "<your user id on the platform>"

    [projects.agent]
    type = "claudecode"

    [projects.agent.options]
    work_dir = "/home/you/.cc-hub"     # absolute path; ~ expansion needs cc-connect > 1.5.1
    mode = "default"                    # approvals are relayed to the chat

    [projects.agent.options.env]
    PATH = "/home/you/.local/bin:/usr/local/bin:/usr/bin:/bin"
    PYTHONUTF8 = "1"

    [[projects.platforms]]
    type = "feishu"

    [projects.platforms.options]
    app_id = "${FEISHU_APP_ID}"
    app_secret = "${FEISHU_APP_SECRET}"
    allow_from = "*"

For Codex use `type = "codex"` with the same `work_dir` and `env`, and read
[codex-sandbox.md](codex-sandbox.md).

## PATH under a daemon

`cc-connect daemon install` runs cc-connect from systemd, launchd or the
Windows Task Scheduler. Those do not read your shell profile, so PATH is often
just the system directories and `cc-hub`, `uv`, `claude` or `codex` are not
found. Put the full PATH in `[projects.agent.options.env]` as above.
`cc-hub doctor` reports which binaries would be missing (`daemon-path`), and
`ENV.md` lists absolute paths the agent can fall back to.

## What the agent uses

- Attachments you send land in `<hub>/.cc-connect/attachments/<message id>/`;
  cc-connect appends their absolute paths to the prompt.
- The agent sends files back with `cc-connect send --file <absolute path>`
  (`--image`, `--tts`). 50 MB limit.
- Scheduled prompts: `cc-connect cron add --cron "<expr>" --prompt "<text>"`.
  The hub keeps definitions in `cron/*.md`; bootstrap registers the defaults.
- `/dir <path>` in the chat switches the working directory for the *next*
  session only. Keep `work_dir` on the hub and let the agent `cd` or use jobs.

## Approvals

`mode = "default"` relays every tool approval to the chat. The hub's hook is
the floor beneath that: it blocks a short list of catastrophic commands even
when you switch to `/mode yolo` or answer "allow all".
```

`docs/windows.md`:

```markdown
# Windows notes

## Recommended: WSL2

Run cc-connect, the agent CLIs and the hub inside WSL2. Keep the hub on the
Linux filesystem (`~/.cc-hub`), not under `/mnt/c`, which is slow and has
different permission semantics. Everything in this project behaves as on Linux.

## Native Windows

cc-hub works natively, with these differences:

- **Links.** Creating symlinks needs Developer Mode or administrator rights,
  so `cc-hub link` creates a directory junction instead; if that fails it
  copies and leaves a `.cc-hub-copied` marker. `cc-hub doctor` reports
  `link-stale` when the copy is older than `skills/`; re-run the `hub-adapt`
  skill to refresh it.
- **Shell.** Claude Code uses Git Bash for its Bash tool when Git for Windows
  is installed and PowerShell otherwise; Codex uses PowerShell. The blocklist
  covers both syntaxes (`rm -rf ~` and `Remove-Item -Recurse -Force`).
- **Hooks.** Use the absolute path printed by `cc-hub hook snippet`; it avoids
  `.cmd` shim problems. Codex hooks accept a `commandWindows` field if you need
  a different command on Windows.
- **Encoding.** All cc-hub output is UTF-8. Job logs are stored as raw bytes
  and decoded on read, so GBK output from legacy tools is still readable.
- **Line endings.** Keep hub files LF. Some skill loaders silently drop
  frontmatter when a `SKILL.md` has CRLF endings.
- **Paths.** The hub defaults to `%USERPROFILE%\.cc-hub`. cc-connect's
  `run_as_user` is not available on Windows; use `mode = "default"` approvals.
```

```markdown
# Codex sandbox notes

cc-connect starts Codex with `codex exec --cd <hub> --sandbox workspace-write`
by default. Inside that sandbox only the working directory (the hub) and
`writable_roots` are writable, and the restriction is inherited by every
child process, including background jobs started with `cc-hub jobs start`.

## Make registered projects writable

Add their paths to `writable_roots` in `.codex/config.toml` inside the hub
(the hub is the Codex project directory, so this file applies once the hub
is trusted):

    [sandbox_workspace_write]
    writable_roots = ["/home/you/dev/project-a", "/home/you/dev/project-b"]

The `hub-adapt` skill writes this from `projects.toml`; run it again after
`cc-hub projects add`. If Codex rejects the key at project level, put the same
table in `~/.codex/config.toml` instead.

Alternative: pass `--add-dir <path>` through cc-connect's agent options for
each project.

## One-time trust

Run `codex` interactively once in the hub, trust the hooks via `/hooks`, and
mark the directory trusted. Without trust Codex ignores `.codex/hooks.json`
and `.codex/config.toml`, so the blocklist hook would not run.

## Full access

`--dangerously-bypass-approvals-and-sandbox` (cc-connect `mode = "yolo"`)
removes the sandbox entirely. The hub's hook still runs and still blocks the
catastrophic commands on its list.
```

```markdown
# Manual acceptance checklist

Environment: WSL2, cc-connect running as a daemon, Claude Code as the agent,
one chat platform (Feishu or any other).

1. `uv tool install cc-hub`, `cc-hub init`, `cc-hub doctor`: everything green
   except `bootstrap-pending` and `pointers`.
2. Configure cc-connect `work_dir` and `env.PATH`, restart the daemon.
3. Send the first message. The agent runs bootstrap: asks for a name, writes
   `IDENTITY.md` and `USER.md`, runs `hub-adapt`, registers the cron task,
   deletes `BOOTSTRAP.md`.
4. `cc-hub doctor`: all green.
5. Ask the agent to remember something. `memory/` has today's file;
   `cc-hub memory search` finds it.
6. Ask the agent to run the tests of a registered project in the background.
   You receive a job id; asking for progress makes the agent use `jobs status`;
   after completion the memory log has a summary.
7. Ask the agent to run a command with Chinese characters and quotes on a
   registered remote host. Output is correct.
8. Ask the agent to run a command that hits the blocklist but is harmless even
   if the hook failed, for example `git push --force origin main` in a
   throwaway repository with no remote. It is blocked and the agent explains.
9. Ask the agent to send a file back to the chat. It arrives.
10. Bump the package version and run `cc-hub update`: `AGENTS.md` changes,
    `SOUL.md`, `USER.md` and `memory/` do not, and `.cc-hub/backups/` holds
    the previous version.
```
